% ===============================
% LAYOUT PATCHES (moderncv fancy / body style v)
% ===============================
% Why: moderncv sets every \cvitem/\cventry text in a p-column of a tabular. Inline
% \raggedright or \hyphenpenalty inside the text argument therefore has no reliable effect.
% Patch the column type instead.
\usepackage{ragged2e}
% ragged2e re-applies these lengths on every font change (e.g. \textbf), so set them
% globally here rather than inside the cell.
\setlength{\RaggedRightRightskip}{0pt plus 3.5em}    % moderate ragged edge
\setlength{\RaggedRightParfillskip}{0pt plus 7cm}    % last line at least ~half full: no one-word last lines
\makeatletter
% main column: ragged-right, no hyphenation (narrow column + justification = rivers and "Informati-onsextraktion")
\expandafter\patchcmd\csname\@backslashchar @cvitem\endcsname{p{\maincolumnwidth}}{>{\RaggedRight\hyphenpenalty=10000\exhyphenpenalty=10000\arraybackslash}p{\maincolumnwidth}}{}{\GenericWarning{}{RAGGED PATCH FAILED}}
% uniform row spacing: fixed depth on the last line of every cvitem cell
\expandafter\patchcmd\csname\@backslashchar @cvitem\endcsname{#3\\}{#3\ifhmode\@moderncvstrut{5pt}{0pt}\fi\\}{}{\GenericWarning{}{CVITEM PATCH FAILED}}
% lists (itemize) reset \parfillskip in \@trivlist; re-apply the balanced last line inside bullets
\apptocmd{\@trivlist}{\parfillskip\RaggedRightParfillskip}{}{\GenericWarning{}{TRIVLIST PATCH FAILED}}
% entry headings: bold title on line 1 next to the date, employer + city on line 2.
% (The stock heading wraps "title, company, city." into the date box and strands the city alone on a line.)
\renewcommand*{\cventry}[7][.5ex]{%
  \savebox{\cventryyearbox}{\hspace*{2\separatorcolumnwidth}\hintstyle{#2}}%
  \setlength{\cventrytitleboxwidth}{\widthof{\usebox{\cventryyearbox}}}%
  \setlength{\cventrytitleboxwidth}{\maincolumnwidth-\cventrytitleboxwidth}%
  \cvitem[#1]{}{%
    \parbox[t]{\cventrytitleboxwidth}{\RaggedRight\strut{\bfseries#3}%
      \ifthenelse{\equal{#4}{}}{}{\newline{\slshape#4}\ifthenelse{\equal{#5}{}}{}{, #5}}\strut}%
    \usebox{\cventryyearbox}}%
  \ifx&#7&\else\vspace*{-4pt}\cvitem[#1]{}{\small#7}\fi}
\makeatother

%    b) Professional Summary: 2-3 sentences emphasizing most relevant aspects
%    c) Core Competencies: Choose 3-4 areas that match job requirements

%    e) Projects: Select 2-4 projects that best demonstrate required skills
%    f) Technologies: Prioritize and organize based on job description

%    - Use similar terminology (e.g., "MLOps" vs "ML Infrastructure")

%
% FOR ML ENGINEER ROLES:
% - Emphasize: System architecture, model development, production ML systems
% - Highlight: Technical depth, scalability, distributed systems
% - Technologies: ML frameworks, serving platforms, experimentation tools
% - Projects: End-to-end ML system design, model optimization
%
% FOR MLOPS / PLATFORM ENGINEER ROLES:
% - Emphasize: Infrastructure automation, deployment velocity, reliability
% - Highlight: CI/CD, monitoring, operational metrics (uptime, deployment frequency)
% - Technologies: Kubernetes, cloud platforms, IaC tools, monitoring stack
% - Projects: Platform migrations, infrastructure modernization, automation
%
% FOR ENGINEERING MANAGER / TECHNICAL LEAD ROLES:
% - Emphasize: Team leadership, strategic impact, cross-functional coordination
% - Highlight: Team size, mentoring, process improvements, stakeholder management
% - Technologies: Balance technical depth with breadth
% - Projects: Team achievements, organizational improvements, strategic initiatives
